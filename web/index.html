01
Scale-Invariant Ordinal Formulation

Monocular imagery lacks parallax disparity, rendering absolute metric depth in meters mathematically ill-posed without calibrated camera intrinsics. Continuous relative ordinal ordering (dA < dB) is fundamentally scale- and shift-invariant, generalizing zero-shot across diverse cameras and lens optics.

02
Koschmieder Optical Scattering Prior

Standard visual CNNs treat scenes as unconstrained RGB textures. The PINN formulation couples visual representations with Koschmieder's physical extinction law (I = J • t + A(1 − t)), penalizing non-physical depth predictions and ensuring monotonic distance ordering in hazy landscapes.

03
Multi-Spectral Dynamic Relief (MSX)

Direct colormapping of continuous depth fields tends to blur adjacent foreground objects into a singular visual cluster. MSX unsharp-masking extracts high-frequency RGB luminance gradients (∇Ihighpass) to modulate the thermal palette, preserving distinct object boundaries (e.g., foliage, vehicles, structures).

04
Real-Time Latency & Efficiency

Built upon a lightweight ResNet-18 backbone with a 4-stage residual FPN decoder (~12.4M parameters). Operates with sub-25ms inference latency (>40 FPS on standard RTX GPUs), ensuring feasibility for embedded robotics, autonomous drones, and low-power edge vision.

PINNDepthNet: Dual-Head Physics-Informed Architecture

01. ResNet-18 Encoder Extracts 5 multi-scale feature maps from fine edge textures (C1) to the global sky/horizon (C5).
02. Relative Depth Head Predicts continuous relative depth d(x) ∈ [0, 1] (0 = near/warm, 1 = far/cool).
03. Atmospheric Physics Head Predicts physical airlight vector A and atmospheric attenuation coefficient β.

# PINN-Depth: Physics-Informed Relative Depth Estimation Network
# =========================================================================
import torch
import torch.nn as nn
from typing import Dict, Tuple

from .encoder import ResNetEncoder
from .decoder import FeaturePyramidDecoder


class RelativeDepthHead(nn.Module):
    """
    Relative Depth Prediction Head.
    Maps 32 decoded features to a single-channel depth map in [0.0, 1.0].
    - 0.0 represents the nearest foreground objects (warm colors).
    - 1.0 represents the farthest background or sky horizon (cool colors).
    """
    def __init__(self, in_channels: int = 32):
        super().__init__()
        # Step 1: Smooth 3x3 convolution
        self.conv1 = nn.Conv2d(in_channels, 16, kernel_size=3, padding=1)
        self.act = nn.ELU(inplace=True)
        # Step 2: 1x1 projection to 1 depth channel
        self.conv2 = nn.Conv2d(16, 1, kernel_size=1)
        # Step 3: Sigmoid guarantees relative depth is bounded in [0.0, 1.0]
        self.sigmoid = nn.Sigmoid()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Input: [B, 32, H, W] -> Output: [B, 1, H, W]
        return self.sigmoid(self.conv2(self.act(self.conv1(x))))


class AtmosphericPhysicsHead(nn.Module):
    """
    Atmospheric Parameter Estimator.
    Estimates global airlight vector 'A' and scattering coefficient 'beta'
    from the deepest semantic feature C5 [B, 512, H/32, W/32].
    """
    def __init__(self, in_channels: int = 512):
        super().__init__()
        # Global pooling collapses spatial dimensions to a single 1D vector
        self.pool = nn.AdaptiveAvgPool2d(1)
        self.fc = nn.Sequential(
            nn.Linear(in_channels, 64),
            nn.ELU(inplace=True),
            nn.Linear(64, 4),  # [A_red, A_green, A_blue, beta]
        )

    def forward(self, c5: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        feat = self.pool(c5).flatten(1)  # Shape: [B, 512]
        params = self.fc(feat)
        # Airlight A in [0.5, 1.0] (sky/haze is bright outdoor light)
        airlight = 0.5 + 0.5 * torch.sigmoid(params[:, :3]).view(-1, 3, 1, 1)
        # Scattering beta in [0.2, 3.0] (natural outdoor attenuation range)
        beta = 0.2 + 2.8 * torch.sigmoid(params[:, 3:]).view(-1, 1, 1, 1)
        return airlight, beta


class PINNDepthNet(nn.Module):
    """
    Complete Physics-Informed Relative Monocular Depth Network.
    
    Data Flow:
      1. RGB Image [B, 3, H, W]
      2. Encoder (ResNet-18) -> extracts 5 multi-scale feature maps [C1..C5]
      3. Decoder (FPN)       -> combines multi-scale features -> [B, 32, H, W]
      4. Depth Head          -> predicts bounded relative depth [B, 1, H, W]
      5. Physics Head        -> estimates global airlight A and scattering beta
      6. Scattering Layer    -> computes Koschmieder transmission t(x) = exp(-beta * d)
    """
    def __init__(self, backbone: str = "resnet18", pretrained: bool = True):
        super().__init__()
        self.encoder = ResNetEncoder(backbone=backbone, pretrained=pretrained)
        self.decoder = FeaturePyramidDecoder(encoder_channels=self.encoder.channels)
        self.depth_head = RelativeDepthHead(in_channels=self.decoder.out_channels)
        self.physics_head = AtmosphericPhysicsHead(in_channels=self.encoder.channels[-1])

    def forward(self, x: torch.Tensor) -> Dict[str, torch.Tensor]:
        # Step 1: Extract multi-scale visual features from RGB image
        features = self.encoder(x)  # [C1, C2, C3, C4, C5]

        # Step 2: Combine scales with FPN Decoder to restore native image resolution
        decoded = self.decoder(features, target_size=x.shape[-2:])  # [B, 32, H, W]

        # Step 3: Predict bounded relative depth in [0.0, 1.0]
        depth = self.depth_head(decoded)  # [B, 1, H, W]

        # Step 4: Estimate global airlight A and attenuation beta
        airlight, beta = self.physics_head(features[-1])

        # Step 5: Enforce Koschmieder optical scattering transmission
        transmission = torch.exp(-beta * depth)  # t(x) = exp(-beta * d(x))

        return {
            "depth": depth,
            "airlight": airlight,
            "beta": beta,
            "transmission": transmission,
        }

FeaturePyramidDecoder: Multi-Scale Skip Connection Fusion

Deep CNN layers understand what an object is (global semantics), but shallow layers know where its exact boundaries are (sharp contours). The decoder fuses both:

01. 1x1 Skip Projections Compresses encoder lateral skip channels (C1..C4) to match decoder channels.
02. Bilinear 2x Upsampling Progressively enlarges spatial resolution by 2x at each stage (Stride 32 → 16 → 8 → 4 → 2).
03. Channel Fusion Concat Concatenates upsampled deep features with fine-grained edge skips and fuses with 3x3 Conv.
04. Residual Boundary Refinement y = x + F(x) refines spatial contours while preventing vanishing gradients.
Decoder Stage Fusion: F_i = Conv3x3(Concat[Upsample_2x(D_{i+1}), Proj(C_i)])  ⇒  D_i = F_i + ResidualConv(F_i)

# =========================================================================
# Multi-Scale FPN Decoder with Residual Refinement
# =========================================================================
import torch
import torch.nn as nn
import torch.nn.functional as F
from typing import List


class ResidualBlock(nn.Module):
    """
    Residual Refinement Block:
    output = input + Conv3x3(ELU(BN(Conv3x3(ELU(BN(input))))))
    """
    def __init__(self, channels: int):
        super().__init__()
        self.conv1 = nn.Conv2d(channels, channels, kernel_size=3, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(channels)
        self.act = nn.ELU(inplace=True)
        self.conv2 = nn.Conv2d(channels, channels, kernel_size=3, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(channels)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.act(self.bn2(self.conv2(self.act(self.bn1(self.conv1(x))))) + x)


class DecoderStage(nn.Module):
    """
    Individual FPN Decoder Stage:
      1. Project skip feature with 1x1 Conv
      2. Upsample deep feature 2x using Bilinear Interpolation
      3. Concatenate and fuse channels with 3x3 Conv
      4. Refine object boundaries with ResidualBlock
    """
    def __init__(self, in_channels: int, skip_channels: int, out_channels: int):
        super().__init__()
        self.skip_proj = nn.Sequential(
            nn.Conv2d(skip_channels, out_channels, kernel_size=1, bias=False),
            nn.BatchNorm2d(out_channels),
            nn.ELU(inplace=True),
        )
        self.fuse_conv = nn.Sequential(
            nn.Conv2d(in_channels + out_channels, out_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(out_channels),
            nn.ELU(inplace=True),
        )
        self.refine = ResidualBlock(out_channels)

    def forward(self, x: torch.Tensor, skip: torch.Tensor) -> torch.Tensor:
        # Step 1: Project skip channels with 1x1 Conv
        skip_feat = self.skip_proj(skip)
        # Step 2: 2x Bilinear upsampling to match skip feature size
        x_up = F.interpolate(x, size=skip_feat.shape[-2:], mode="bilinear", align_corners=True)
        # Step 3: Concatenate and fuse channels
        fused = self.fuse_conv(torch.cat([x_up, skip_feat], dim=1))
        # Step 4: Refine boundaries
        return self.refine(fused)


class FeaturePyramidDecoder(nn.Module):
    """
    Hierarchical Decoder: Fuses C5 down through C1 to native resolution [B, 32, H, W].
    """
    def __init__(self, encoder_channels: List[int] = [64, 64, 128, 256, 512]):
        super().__init__()
        self.out_channels = 32
        c1, c2, c3, c4, c5 = encoder_channels

        # Compress top semantic feature C5 (512 channels) -> 128
        self.top_proj = nn.Sequential(
            nn.Conv2d(c5, 128, 1, bias=False),
            nn.BatchNorm2d(128),
            nn.ELU(inplace=True),
            ResidualBlock(128),
        )
        # Progressive top-down decoding stages
        self.stage4 = DecoderStage(128, c4, 128)  # Stride 16
        self.stage3 = DecoderStage(128, c3, 128)  # Stride 8
        self.stage2 = DecoderStage(128, c2, 128)  # Stride 4
        self.stage1 = DecoderStage(128, c1, 64)   # Stride 2

        # Final stage: native resolution H x W with 32 output channels
        self.final_stage = nn.Sequential(
            nn.Conv2d(64, 32, 3, padding=1, bias=False),
            nn.BatchNorm2d(32),
            nn.ELU(inplace=True),
            ResidualBlock(32),
        )

    def forward(self, features: List[torch.Tensor], target_size=None) -> torch.Tensor:
        c1, c2, c3, c4, c5 = features
        d5 = self.top_proj(c5)
        d4 = self.stage4(d5, c4)
        d3 = self.stage3(d4, c3)
        d2 = self.stage2(d3, c2)
        d1 = self.stage1(d2, c1)
        # Final bilinear upsample to input image size
        out = F.interpolate(d1, size=target_size, mode="bilinear", align_corners=True)
        return self.final_stage(out)

Physics-Informed Multi-Task Loss Objectives

Rather than requiring expensive LiDAR laser scans, our network is supervised with 3 self-consistent loss terms:

01. Pairwise Ranking Loss (L_rank) Samples pairs of points (A, B) and penalizes ordering violations whenever Point A is closer than Point B.
02. Koschmieder Consistency (L_phys) Reconstructs intrinsic radiance J(x) and ensures the scattering equation matches observed scene I(x).
03. Edge-Aware Smoothness (L_smooth) Enforces smooth depth across flat surfaces while allowing sharp depth cliffs at true object silhouettes.
Total Training Loss: L_{Total} = L_{ranking}(d) + λ_{phys} ||I - [J • t(d) + A(1 - t(d))]||_1 + λ_{smooth} L_{edge}(d, I)

# =========================================================================
# Physics-Informed Relative Depth Losses
# =========================================================================
import torch
import torch.nn as nn
import torch.nn.functional as F


class RelativeDepthRankingLoss(nn.Module):
    """
    Pairwise Ordinal Ranking Loss.
    Enforces correct relative ordering between pairs of pixels:
        If Point A is in front of Point B (depth_A < depth_B),
        penalize any prediction where depth_A >= depth_B.
    """
    def __init__(self, num_pairs: int = 5000, margin_tau: float = 0.03):
        super().__init__()
        self.num_pairs = num_pairs
        self.margin_tau = margin_tau

    def forward(self, pred: torch.Tensor, gt: torch.Tensor, mask: torch.Tensor = None) -> torch.Tensor:
        B, _, H, W = pred.shape
        if mask is None:
            mask = (gt > 0) & torch.isfinite(gt)

        total_loss = torch.tensor(0.0, device=pred.device)
        for b in range(B):
            indices = torch.nonzero(mask[b, 0])
            if len(indices) < 100: continue

            # Step 1: Sample random point pairs
            N = min(self.num_pairs, len(indices) // 2)
            i0 = torch.randint(0, len(indices), (N,))
            i1 = torch.randint(0, len(indices), (N,))

            # Step 2: Compute ground-truth and predicted differences
            diff_gt = gt[b, 0, indices[i0, 0], indices[i0, 1]] - gt[b, 0, indices[i1, 0], indices[i1, 1]]
            diff_pr = pred[b, 0, indices[i0, 0], indices[i0, 1]] - pred[b, 0, indices[i1, 0], indices[i1, 1]]

            # Step 3: Target ordinal ordering label
            labels = torch.zeros_like(diff_gt)
            labels[diff_gt > self.margin_tau] = 1.0
            labels[diff_gt < -self.margin_tau] = -1.0

            # Step 4: Softplus ranking loss
            ineq = labels != 0
            if ineq.any():
                total_loss += torch.log1p(torch.exp(-labels[ineq] * diff_pr[ineq])).mean()
        return total_loss / max(1, B)


class PhysicsConsistencyLoss(nn.Module):
    """
    Koschmieder Physical Consistency Loss:
    Reconstructs the image and penalizes L1 error: ||I - [J*t + A*(1 - t)]||_1
    """
    def forward(self, image, pred_depth, airlight, beta) -> torch.Tensor:
        # Step 1: Koschmieder transmission
        t = torch.exp(-beta * pred_depth)
        # Step 2: Recover scene radiance
        j = torch.clamp((image - airlight) / torch.clamp(t, 1e-4) + airlight, 0.0, 1.0)
        # Step 3: Reconstruct image using optical scattering model
        reconstructed = j * t + airlight * (1.0 - t)
        # Step 4: L1 reconstruction error
        return F.l1_loss(reconstructed, image)

FLIR MSX Detail Embossing & Contrast Enhancement

Standard thermal colormaps can blur adjacent objects into a monochromatic cluster. We apply two optical post-processing algorithms:

01. High-Pass Spatial Filter Extracts fine edge textures (flower petals, tree bark, vehicle outlines) via Gaussian unsharp masking.
02. Dynamic MSX Relief Embossing Embosses the physical edge relief directly onto the thermal depth map while strictly preserving color hues.
03. Percentile Dynamic Tone Stretch Normalizes between 1st and 99th percentiles with 1.35x gamma to maximize depth separation between foreground and background.
MSX Relief Equation: H(x) = Y_{gray}(x) − GaussianBlur(Y_{gray}, σ=1.3)  ⇒  C_{fused} = C_{thermal} • [(1 - s) + 2s(0.5 + 2.4 H(x))]

# =========================================================================
# FLIR MSX Dynamic Detail Embossing & Tone Expansion
# =========================================================================
import cv2
import numpy as np


def apply_flir_msx_detail_embossing(
    thermal_rgb: np.ndarray,
    img_rgb: np.ndarray,
    strength: float = 0.55,
) -> np.ndarray:
    """
    Extracts high-frequency visual edge relief from RGB luminance and
    embosses it directly onto the thermal depth map, achieving authentic FLIR MSX clarity.
    """
    if strength <= 0.001:
        return thermal_rgb

    # Step 1: Extract grayscale luminance from visible RGB image
    gray = cv2.cvtColor(img_rgb, cv2.COLOR_RGB2GRAY).astype(np.float32) / 255.0

    # Step 2: High-pass spatial edge filter via Gaussian unsharp masking
    blur = cv2.GaussianBlur(gray, (0, 0), sigmaX=1.3)
    highpass = gray - blur

    # Step 3: Compute edge relief modulation factor
    edge_relief = np.clip(0.5 + 2.4 * highpass, 0.0, 1.0)

    # Step 4: Modulate thermal pixel brightness while strictly preserving color hues
    thermal_float = thermal_rgb.astype(np.float32) / 255.0
    modulator = (1.0 - strength) + (2.0 * strength) * edge_relief[:, :, np.newaxis]
    fused = np.clip(thermal_float * modulator, 0.0, 1.0)
    return (fused * 255.0).astype(np.uint8)


def enhance_depth_contrast(depth: np.ndarray, gamma: float = 1.35) -> np.ndarray:
    """
    Stretches active percentile dynamic range (1% to 99%) and applies
    gamma expansion to separate midground objects from background.
    """
    # Step 1: Robust percentile normalization (rejects outlier spikes)
    p1, p99 = np.percentile(depth, 1.0), np.percentile(depth, 99.0)
    stretched = np.clip((depth - p1) / max(p99 - p1, 1e-5), 0.0, 1.0)

    # Step 2: Non-linear midtone contrast expansion
    return np.power(stretched, gamma)
